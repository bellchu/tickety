  kubectl get service -n {{ .Release.Namespace }} {{ include "tickety.fullname" . }}-frontend --watch
{{- else if .Values.ingress.enabled }}

  kubectl port-forward -n {{ .Release.Namespace }} service/{{ include "tickety.fullname" . }}-frontend 3000:{{ .Values.frontend.service.port }}

{{- end }}

Check the rollout:
  kubectl get pods,jobs -n {{ .Release.Namespace }}
  kubectl get --raw /api/v1/namespaces/{{ .Release.Namespace }}/services/http:{{ include "tickety.fullname" . }}-frontend:{{ .Values.frontend.service.port }}/proxy/api/health/ready
